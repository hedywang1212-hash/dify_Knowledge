### 一、jelly基本语法

1.if
```
test后面写判断条件
<j:if test="${!jvar_XXX}">
   ${jvar_XXX}为false时执行
</j:if>
<j:if test="${jvar_XXX}">
	${jvar_XXX}为true时执行
</j:if>
```
2.while
```
执行while循环
<j:while test="${jvar_xxx}">
</j:while>
```
3.set
```
//设置一个变量jvar_incident_number，值为${jvar_now_GR.getValue('number')}
<j:set var="jvar_incident_number" value="${jvar_now_GR.getValue('number')}"/>
```
4.set_if
`<g:set_if var="jvar_XXX" test="<判断条件>" true="条件为true时的值" false="条件为false时的值" />`
5.choose
```
相当于 ifelse
<j:choose>
   <j:when test="${jvar_now_GR.getRowCount() ${AMP}lt; 1}">We found multiple records!</j:when>
   <j:when test="${jvar_now_GR.next()}">We found record ${jvar_now_GR.getValue('number')}</j:when>
   <j:otherwise>Sorry, we could not find the record you specified.</j:otherwise>
</j:choose>
```
6.e v a luate
可写服务器端js代码
| 字段       | 说明                                     |
| ---------- | ---------------------------------------- |
| var        | 定义变量                                 |
| object     | 值为true时，表达式的结果将被视为一个对象 |
| jelly      | 值为true时，允许脚本引用jelly上下文变量  |
| expression | 表达式的最后一行时传递给var的实际值      |
```
<g:e v a luate var="jvar_now_GR" object="true">
  var now_GR = new GlideRecord("incident");
  now_GR.addQuery("active", "true");
  now_GR.query();
  now_GR; // this is the variable put into the variable jvar_now_GR
</g:e v a luate>

或者

<g:e valuate var="jvar_now_GR" object="true" expression="
  var now_GR = new GlideRecord('incident');
  now_GR.addQuery('active', 'true');
  now_GR.query();
  now_GR; // this is the variable put into the variable jvar_now_GR" />
```

7.breakpoint
```
断点打印出当前时刻jelly中所有变量的列表，以及各自的值。也可以通过var指定打印的变量
<g:breakpoint />
<g:breakpoint var="sysparm_view"/>
```
8.macro_invoke
```
调用一个name为sample_macro的ui macro，并且定义一个jvar_变量 赋值为变量值.
<g:macro_invoke macro="sample_macro" <变量>="<变量值>" />
或者
<g:sample_macro <变量>="<变量值>" />
```



### 二、<g>和<g2>的区别  <j>和<j2>的区别
1.<g>和<j>输出会被缓存，<g2>和<j2>每次页面显示时，会重新计算其值
2.<g>、<j>和${}搭配使用，<g2>、<j2>和$[]使用



###   三、注意点
##### 1.jelly的javascript中不可直接使用${jvar_xxx}或$[jvar_xxx]来获取值，但是可以使用以下方法来获取

方法1：
```
<script>
var a="${jvar_xxx}"`
</script>
//变量a将获得jvar_xxx的值
```

方法2：通过函数传参

`<button o n c l i c k="myFunction('jvar_xxx')"></button>`

#### 2.当ui macro的name(g:macro)是ui macro的名字时,使用${jvar_name}可获取ui macro的name,并且可以进行拼接字符串。
```
 <g:macro name="test_macro_name" />
<div>${jvar_name}<div>   //此处会显示test_macro_name
 function ${jvar_name}_alert() {
    alert("Hello,jelly")
  }
```

```
<?xml version="1.0" encoding="utf-8" ?>
<j:jelly trim="false" xmlns:j="jelly:core" xmlns:g="glide" xmlns:j2="null" xmlns:g2="null">
<g:e valuate var="jvar_guid" expression="gs.generateGUID(this);" />

<a id="${jvar_n}" 
 o n c l i c k="openOutboundCall();"
  name="${jvar_n}" 
  tabindex="0"
  style="display:none"
  class="reference_decoration btn btn-default"
  title="${gs.getMessage('Outbound Call')}"
>
<img border="0" src="images/icons/phone.gif" title="${gs.getMessage('Outbound Call')}" alt="${gs.getMessage('Outbound Call')}"/>
</a>
 
<script>
function openOutboundCall(){
var baseurl = "cti://agent?phone=";
var caller = g_form.getValue('u_called_phone_number');
window.open(baseurl + caller);
}
 
function decorationShow(element, original, changed, loading) {
var $refButton = $j(gel('${jvar_n}'));
if (changed.length > 0)
$refButton.show().children().show();
else
$refButton.hide();
}
 
var n = '${ref}'.replace(/\./g, '_');
var h = new GlideEventHandler('onChange_' + n, decorationShow, '${ref}');
g_event_handlers.push(h);
 
</script>
</j:jelly>
```
